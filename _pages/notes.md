  (function () {
    var PASSWORD_HASH = "0d3cb1b21734367f65d267b902b3122a1afcb108ec0405f6c8d53473c2fab48a";
    var gate = document.getElementById("notes-gate");
    var content = document.getElementById("notes-content");
    var input = document.getElementById("notes-password");
    var button = document.getElementById("notes-unlock");
    var error = document.getElementById("notes-error");

    function sha256Hex(text) {
      var data = new TextEncoder().encode(text);
      return crypto.subtle.digest("SHA-256", data).then(function (buf) {
        return Array.from(new Uint8Array(buf))
          .map(function (b) { return b.toString(16).padStart(2, "0"); })
          .join("");
      });
    }

    function reveal() {
      gate.hidden = true;
      content.hidden = false;
    }

    function unlock() {
      sha256Hex(input.value).then(function (hash) {
        if (hash === PASSWORD_HASH) {
          try { sessionStorage.setItem("notes-unlocked", "1"); } catch (e) {}
          reveal();
        } else {
          error.hidden = false;
        }
      });
    }

    button.addEventListener("click", unlock);
    input.addEventListener("keydown", function (e) {
      if (e.key === "Enter") unlock();
    });

    try {
      if (sessionStorage.getItem("notes-unlocked") === "1") {
        reveal();
      }
    } catch (e) {}
  })();
</script>
